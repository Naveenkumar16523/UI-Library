# Dock

An interactive navigation dock that magnifies icons smoothly as you hover over them, perfect for global navigation on dashboards or premium web apps.

## Usage

```tsx
import { Dock, DockIcon } from "@/components/ui/dock";

export default function DockDemo() {
  return (
    <Dock>
      <DockIcon>🏠</DockIcon>
      <DockIcon>🔍</DockIcon>
      <DockIcon>⚙️</DockIcon>
    </Dock>
  );
}
```
