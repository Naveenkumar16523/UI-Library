# Text Shimmer

A subtle, repeating light sweep effect across text. Perfect for inline badges, "New" announcements, or premium upsell copy.

## Usage

```tsx
import { TextShimmer } from "@/components/ui/text-shimmer";

export default function ShimmerDemo() {
  return (
    <TextShimmer className="text-xl font-bold">
      ✨ Introducing AI Features
    </TextShimmer>
  );
}
```
